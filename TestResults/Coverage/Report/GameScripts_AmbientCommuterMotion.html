 406    static void EnsurePostersBuilt()
 407    {
 408        if (cachedPosterMaterials != null && cachedPosterMaterials.Length == PosterCount)
 409            return;
 410
 411        cachedTextures = new Texture2D[PosterCount];
 412        cachedPosterMaterials = new Material[PosterCount];
 413
 414        Shader shader = BuiltInShaders.Lit;
 415
 416        for (int i = 0; i < PosterCount; i++)
 417        {
 418            var tex = BuildPosterTexture(i);

 421            var mat = new Material(shader) { name = $"BusStop_PosterMat_{i}" };
 422            mat.mainTexture = tex;
 423            if (mat.HasProperty("_MainTex")) mat.SetTexture("_MainTex", tex);
 424            if (mat.HasProperty("_EmissionColor"))
 425            {
 426                mat.EnableKeyword("_EMISSION");
 427                mat.SetColor("_EmissionColor", new Color(0.18f, 0.18f, 0.18f));
 428                if (mat.HasProperty("_EmissionMap")) mat.SetTexture("_EmissionMap", tex);
 429            }
 430            cachedPosterMaterials[i] = mat;
 431        }
 432    }

 434    static Texture2D BuildPosterTexture(int index)
 435    {
 436        var tex = new Texture2D(PosterWidth, PosterHeight, TextureFormat.RGBA32, false)
 437        {
 438            wrapMode = TextureWrapMode.Clamp,
 439            filterMode = FilterMode.Bilinear,
 440            name = $"Poster_{index}"
 441        };
 442
 443        var pixels = new Color32[PosterWidth * PosterHeight];
 444        var ctx = new CanvasContext(pixels, PosterWidth, PosterHeight);
 445
 446        switch (index)
 447        {
 448            case 0: BuildPoster_MetroRapid(ctx); break;
 449            case 1: BuildPoster_SparkCola(ctx); break;
 450            case 2: BuildPoster_CafeAurora(ctx); break;
 451            case 3: BuildPoster_SkylineSuites(ctx); break;
 452            case 4: BuildPoster_RealbusCare(ctx); break;
 453            case 5: BuildPoster_SunsetFestival(ctx); break;
 454            case 6: BuildPoster_BurgerCrunch(ctx); break;
 455            default: BuildPoster_Pulse5G(ctx); break;
 456        }
 457
 458        tex.SetPixels32(pixels);
 459        tex.Apply(false, false);
 460        return tex;
 461    }

 663    // Drawing context helper class. Internal rather than private so the street
 664    // advertisement library can use the same pixel font and drawing primitives
 665    // without carrying a second copy of the glyph table.
 666    internal class CanvasContext
 667    {
 668        readonly Color32[] pixels;
 669        readonly int width;
 670        readonly int height;
 671
 672        public int Width => width;
 673        public int Height => height;
 674
 675        public CanvasContext(Color32[] pixels, int width, int height)
 676        {
 677            this.pixels = pixels;
 678            this.width = width;
 679            this.height = height;
 680        }
 681
 682        public void FillRect(int x, int y, int w, int h, Color32 c)
